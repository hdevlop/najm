In this tutorial, you'll build a **Task Management API** with:

- ✅ RESTful CRUD operations
- ✅ PostgreSQL database with Drizzle ORM
- ✅ JWT authentication
- ✅ Request validation
- ✅ Authorization guards
- ✅ Event-driven notifications

## What You'll Build

By the end of this tutorial, you'll have a fully functional API with these endpoints:

```
POST   /api/auth/register    # Create account
POST   /api/auth/login       # Get JWT token
GET    /api/tasks            # List all tasks (authenticated)
POST   /api/tasks            # Create task (authenticated)
GET    /api/tasks/:id        # Get task by ID (authenticated)
PUT    /api/tasks/:id        # Update task (authenticated, owner only)
DELETE /api/tasks/:id        # Delete task (authenticated, owner only)
```

## Step 5: Create Database Schema

Create `src/shared/database/schema.ts`:

```typescript
import { pgTable, text, timestamp, uuid, boolean } from 'drizzle-orm/pg-core';

export const users = pgTable('users', {
  id: uuid('id').primaryKey().defaultRandom(),
  name: text('name').notNull(),
  email: text('email').notNull().unique(),
  password: text('password').notNull(),
  createdAt: timestamp('created_at').defaultNow(),
});

export const tasks = pgTable('tasks', {
  id: uuid('id').primaryKey().defaultRandom(),
  title: text('title').notNull(),
  description: text('description'),
  completed: boolean('completed').default(false),
  userId: uuid('user_id').references(() => users.id).notNull(),
  createdAt: timestamp('created_at').defaultNow(),
  updatedAt: timestamp('updated_at').defaultNow(),
});
```

Create `src/shared/database/connection.ts`:

## Step 8: Create Auth Guard

Create `src/modules/auth/auth.guard.ts`:

```typescript
import { Injectable, DI } from 'najm-di';
import { Headers } from 'najm-params';
import { HttpError } from 'najm-core';
import { Container, USER } from 'najm-core';
import jwt from 'jsonwebtoken';

@Injectable()
export class AuthGuard {
  @DI() container!: Container;

  async canActivate(@Headers('authorization') authHeader: string) {
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      throw new HttpError(401, 'Missing or invalid authorization header');
    }

    const token = authHeader.substring(7);

    try {
      const payload = jwt.verify(token, process.env.JWT_SECRET!) as {
        userId: string;
        email: string;
      };

      // Store user in request context
      this.container.set(USER, payload);

      return payload;
    } catch (error) {
      throw new HttpError(401, 'Invalid or expired token');
    }
  }
}
```

## Step 9: Create Auth Module

Create `src/modules/auth/auth.service.ts`:

```typescript
import { Service } from 'najm-di';
import { DB } from 'najm-database';
import { HttpError } from 'najm-core';
import type { Database } from 'drizzle-orm';
import { users } from '../../shared/database/schema';
import { eq } from 'drizzle-orm';
import bcrypt from 'bcrypt';
import jwt from 'jsonwebtoken';
import type { RegisterDto, LoginDto } from './auth.dto';

@Service()
export class AuthService {
  @DB('default') db!: Database;

  async register(dto: RegisterDto) {
    // Check if user exists
    const existing = await this.db.query.users.findFirst({
      where: eq(users.email, dto.email),
    });

    if (existing) {
      throw new HttpError(400, 'Email already registered');
    }

    // Hash password
    const hashedPassword = await bcrypt.hash(dto.password, 10);

    // Create user
    const [user] = await this.db
      .insert(users)
      .values({
        name: dto.name,
        email: dto.email,
        password: hashedPassword,
      })
      .returning();

    // Generate token
    const token = this.generateToken(user.id, user.email);

    return {
      user: {
        id: user.id,
        name: user.name,
        email: user.email,
      },
      token,
    };
  }

  async login(dto: LoginDto) {
    // Find user
    const user = await this.db.query.users.findFirst({
      where: eq(users.email, dto.email),
    });

    if (!user) {
      throw new HttpError(401, 'Invalid credentials');
    }

    // Verify password
    const valid = await bcrypt.compare(dto.password, user.password);
    if (!valid) {
      throw new HttpError(401, 'Invalid credentials');
    }

    // Generate token
    const token = this.generateToken(user.id, user.email);

    return {
      user: {
        id: user.id,
        name: user.name,
        email: user.email,
      },
      token,
    };
  }

  private generateToken(userId: string, email: string): string {
    return jwt.sign(
      { userId, email },
      process.env.JWT_SECRET!,
      { expiresIn: process.env.JWT_EXPIRES_IN || '7d' }
    );
  }
}
```

Create `src/modules/auth/auth.controller.ts`:

```typescript
import { Controller } from 'najm-di';
import { Post } from 'najm-router';
import { Body } from 'najm-params';
import { Validate } from 'najm-validation';
import { AuthService } from './auth.service';
import { registerSchema, loginSchema, RegisterDto, LoginDto } from './auth.dto';

@Controller('/api/auth')
export class AuthController {
  constructor(private authService: AuthService) {}

  @Post('/register')
  @Validate(registerSchema)
  async register(@Body() dto: RegisterDto) {
    return this.authService.register(dto);
  }

  @Post('/login')
  @Validate(loginSchema)
  async login(@Body() dto: LoginDto) {
    return this.authService.login(dto);
  }
}
```

## Step 10: Create Tasks Module

Create `src/modules/tasks/task.service.ts`:

```typescript
import { Service, Events } from 'najm-di';
import { DB } from 'najm-database';
import { HttpError } from 'najm-core';
import type { Database } from 'drizzle-orm';
import { tasks } from '../../shared/database/schema';
import { eq, and } from 'drizzle-orm';
import type { CreateTaskDto, UpdateTaskDto } from './task.dto';

@Service()
export class TaskService {
  @DB('default') db!: Database;
  @Events() events!: { emit: (event: string, data: any) => void };

  async findAllByUser(userId: string) {
    return this.db.query.tasks.findMany({
      where: eq(tasks.userId, userId),
      orderBy: (tasks, { desc }) => [desc(tasks.createdAt)],
    });
  }

  async findById(id: string, userId: string) {
    const task = await this.db.query.tasks.findFirst({
      where: and(eq(tasks.id, id), eq(tasks.userId, userId)),
    });

    if (!task) {
      throw new HttpError(404, 'Task not found');
    }

    return task;
  }

  async create(dto: CreateTaskDto, userId: string) {
    const [task] = await this.db
      .insert(tasks)
      .values({
        ...dto,
        userId,
      })
      .returning();

    this.events.emit('task.created', { taskId: task.id, userId });

    return task;
  }

  async update(id: string, dto: UpdateTaskDto, userId: string) {
    // Verify ownership
    await this.findById(id, userId);

    const [updated] = await this.db
      .update(tasks)
      .set({
        ...dto,
        updatedAt: new Date(),
      })
      .where(and(eq(tasks.id, id), eq(tasks.userId, userId)))
      .returning();

    this.events.emit('task.updated', { taskId: id, userId });

    return updated;
  }

  async delete(id: string, userId: string) {
    // Verify ownership
    await this.findById(id, userId);

    await this.db
      .delete(tasks)
      .where(and(eq(tasks.id, id), eq(tasks.userId, userId)));

    this.events.emit('task.deleted', { taskId: id, userId });

    return { message: 'Task deleted successfully' };
  }
}
```

Create `src/modules/tasks/task.controller.ts`:

```typescript
import { Controller } from 'najm-di';
import { Get, Post, Put, Delete } from 'najm-router';
import { Params, Body, User } from 'najm-params';
import { Guards } from 'najm-guard';
import { Validate } from 'najm-validation';
import { AuthGuard } from '../auth/auth.guard';
import { TaskService } from './task.service';
import { createTaskSchema, updateTaskSchema, CreateTaskDto, UpdateTaskDto } from './task.dto';

@Controller('/api/tasks')
@Guards(AuthGuard)
export class TaskController {
  constructor(private taskService: TaskService) {}

  @Get('/')
  async findAll(@User() user: { userId: string }) {
    return this.taskService.findAllByUser(user.userId);
  }

  @Get('/:id')
  async findOne(
    @Params('id') id: string,
    @User() user: { userId: string }
  ) {
    return this.taskService.findById(id, user.userId);
  }

  @Post('/')
  @Validate(createTaskSchema)
  async create(
    @Body() dto: CreateTaskDto,
    @User() user: { userId: string }
  ) {
    return this.taskService.create(dto, user.userId);
  }

  @Put('/:id')
  @Validate(updateTaskSchema)
  async update(
    @Params('id') id: string,
    @Body() dto: UpdateTaskDto,
    @User() user: { userId: string }
  ) {
    return this.taskService.update(id, dto, user.userId);
  }

  @Delete('/:id')
  async delete(
    @Params('id') id: string,
    @User() user: { userId: string }
  ) {
    return this.taskService.delete(id, user.userId);
  }
}
```

## Step 12: Create Server Entry Point

Create `src/index.ts`:

```typescript
import 'reflect-metadata';
import { Server } from 'najm-core';
import { database } from 'najm-database';
import { guards } from 'najm-guard';
import { validation } from 'najm-validation';
import { events } from 'najm-event';
import { db } from './shared/database/connection';

// Auth module
import { AuthController } from './modules/auth/auth.controller';
import { AuthService } from './modules/auth/auth.service';
import { AuthGuard } from './modules/auth/auth.guard';

// Tasks module
import { TaskController } from './modules/tasks/task.controller';
import { TaskService } from './modules/tasks/task.service';
import { TaskEventListener } from './modules/tasks/task.listener';

const server = new Server()
  .use(database({ default: db }))
  .use(guards())
  .use(validation())
  .use(events())
  .load(
    // Auth
    AuthController,
    AuthService,
    AuthGuard,
    // Tasks
    TaskController,
    TaskService,
    TaskEventListener
  );

const port = parseInt(process.env.PORT || '3000');

await server.listen(port, () => {
  console.log(`🚀 Task API running on http://localhost:${port}`);
  console.log(`📚 Endpoints:`);
  console.log(`   POST   /api/auth/register`);
  console.log(`   POST   /api/auth/login`);
  console.log(`   GET    /api/tasks`);
  console.log(`   POST   /api/tasks`);
  console.log(`   GET    /api/tasks/:id`);
  console.log(`   PUT    /api/tasks/:id`);
  console.log(`   DELETE /api/tasks/:id`);
});
```
