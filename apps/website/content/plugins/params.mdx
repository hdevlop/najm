The Parameter system provides 40+ decorators for extracting request data into controller method parameters with automatic parsing and caching.

## Quick Setup

```typescript
import { params } from 'najm-params';
import { Server } from 'najm-core';

const server = new Server()
  .use(params());  // Auto-registered by default
```

## Quick Example

```typescript
import { Body, Params, Query, Headers } from 'najm-params';

@Controller('/products')
class ProductController {
  @Get('/:id')
  getProduct(
    @Params('id') id: string,
    @Query('include') include?: string
  ) {
    return { id, include };
  }

  @Post()
  createProduct(
    @Body() data: CreateProductDto,
    @Headers('authorization') auth: string
  ) {
    return { data, auth };
  }
}
```
